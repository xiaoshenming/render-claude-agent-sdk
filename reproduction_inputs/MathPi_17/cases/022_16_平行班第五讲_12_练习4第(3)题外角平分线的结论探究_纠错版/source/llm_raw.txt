class Solution(BaseConfigScene):
    def construct(self):
        # 1. 原始几何点定义
        C_raw = np.array([0, 0, 0])
        B_raw = np.array([-4, 0, 0])
        E_raw = np.array([-10.31, 0, 0])
        A_raw = np.array([-3.19, 2.23, 0])
        D_raw = np.array([-2.0, 1.40, 0])

        # 2. 创建基础图形
        edge_EC = Line(E_raw, C_raw, color=BASE_COLOR)
        edge_AC = Line(A_raw, C_raw, color=BASE_COLOR)
        edge_AB = Line(A_raw, B_raw, color=BASE_COLOR)
        edge_BD = Line(B_raw, D_raw, color=BASE_COLOR)
        edge_AE = Line(A_raw, E_raw, color=BASE_COLOR)

        diagram = VGroup(edge_EC, edge_AC, edge_AB, edge_BD, edge_AE)
        
        # 适配左侧
        self.fit_left(diagram)

        # 3. 获取适配后的真实坐标
        E = edge_EC.get_start()
        C = edge_EC.get_end()
        A = edge_AC.get_start()
        B = edge_AB.get_end()
        D = edge_BD.get_end()
        F = A + (A - C) * 0.3

        # 4. 创建标签
        lbl_A = MathTex("A", color=TEXT_COLOR).next_to(A, UP, buff=0.15)
        lbl_B = MathTex("B", color=TEXT_COLOR).next_to(B, DOWN, buff=0.15)
        lbl_C = MathTex("C", color=TEXT_COLOR).next_to(C, DOWN, buff=0.15)
        lbl_D = MathTex("D", color=TEXT_COLOR).next_to(D, UR, buff=0.1)
        lbl_E = MathTex("E", color=TEXT_COLOR).next_to(E, DOWN, buff=0.15)

        # 5. 创建辅助线与角标
        aux_AF = DashedLine(A, F, color=DRAW_COLOR)
        arc_BAC = self.make_angle_mark(A, C, B, radius=0.4, color=KNOWN_COLOR)
        arc_ACB = self.make_angle_mark(C, B, A, radius=0.4, color=KNOWN_COLOR)
        arc_ABD = self.make_angle_mark(B, A, D, radius=0.3, color=KNOWN_COLOR)
        arc_DBC = self.make_angle_mark(B, D, C, radius=0.3, color=KNOWN_COLOR)
        arc_FAE = self.make_angle_mark(A, F, E, radius=0.3, color=KNOWN_COLOR)
        arc_EAB = self.make_angle_mark(A, E, B, radius=0.3, color=KNOWN_COLOR)

        # 6. 创建区域色块
        region_ABD = self.make_region_fill(A, B, D, color=PAIR_A_COLOR, opacity=0.18)
        region_ACB = self.make_region_fill(A, C, B, color=PAIR_B_COLOR, opacity=0.18)

        # 7. 整体适配
        geometry_all = VGroup(
            diagram, lbl_A, lbl_B, lbl_C, lbl_D, lbl_E, 
            aux_AF, arc_BAC, arc_ACB, arc_ABD, arc_DBC, arc_FAE, arc_EAB,
            region_ABD, region_ACB
        )
        self.fit_geometry_group(geometry_all)

        # 8. 重新定义用于高亮的边（使用适配后的坐标）
        edge_AB_real = Line(A, B, color=BASE_COLOR)
        edge_BC_real = Line(B, C, color=BASE_COLOR)
        edge_CD_real = Line(C, D, color=BASE_COLOR)
        edge_AD_real = Line(A, D, color=BASE_COLOR)
        edge_BE_real = Line(B, E, color=BASE_COLOR)
        edge_BD_real = Line(B, D, color=BASE_COLOR)
        edge_AE_real = Line(A, E, color=BASE_COLOR)

        # 初始状态：只显示基础三角形和点
        self.add(diagram, lbl_A, lbl_B, lbl_C, lbl_D, lbl_E)

        # ==================== 读题展示阶段 ====================
        with self.voiceover(text="同学们好，我们一起来看这道题。如图，在三角形 ABC 中，角 BAC 等于七十五度，角 ACB 等于三十五度。") as tracker:
            self.play_after_speech([Create(arc_BAC), Create(arc_ACB)], tracker)

        with self.voiceover(text="角 ABC 的平分线 BD 交边 AC 于点 D。") as tracker:
            self.play_after_speech([self.reemphasize_stroke(edge_BD_real), Create(arc_ABD), Create(arc_DBC)], tracker)

        with self.voiceover(text="第三问中，若角 BAC 外角的平分线 AE 交 CB 延长线于点 E，我们需要探究第二问的结论是否仍然成立。") as tracker:
            self.play_after_speech([self.reemphasize_stroke(edge_AE_real), Create(aux_AF), Create(arc_FAE), Create(arc_EAB)], tracker)

        # ==================== 思路切入阶段 ====================
        with self.voiceover(text="我们先来理清思路。首先可以通过角度计算，证明三角形 ABD 与三角形 ACB 相似。") as tracker:
            self.play_steps_evenly([FadeIn(region_ABD), FadeIn(region_ACB)], tracker)

        with self.voiceover(text="接着，利用外角平分线定理，求出 BE 与 AB、AC 的比例关系。") as tracker:
            self.play_steps_evenly([
                self.reemphasize_stroke(edge_AE_real), 
                self.reemphasize_stroke(edge_BE_real), 
                self.reemphasize_stroke(edge_AB_real)
            ], tracker)

        with self.voiceover(text="最后，结合第一问中 BD 等于 DC 的结论，进行线段代换，从而确定最终的关系。") as tracker:
            self.play_steps_evenly([FadeOut(region_ABD), FadeOut(region_ACB)], tracker)

        # ==================== 满分答案阶段 ====================
        # 第一屏
        panel1 = self.make_step_panel([
            self.make_ctex_row(r"解：(2)中的结论不成立，正确关系为 $BE=AB+AD+BD$"),
            self.make_proof_because(r"\angle ABC = 180^\circ - 75^\circ - 35^\circ = 70^\circ"),
            self.make_proof_therefore(r"\angle ABD = 35^\circ = \angle ACB"),
            self.make_proof_therefore(r"\triangle ABD \sim \triangle ACB")
        ])

        with self.voiceover(text="我们来写出严谨的证明。首先，角 ABC 等于一百八十度减去七十五度再减去三十五度，等于七十度。") as tracker:
            self.play_steps_evenly([Write(panel1[0]), Write(panel1[1])], tracker)

        with self.voiceover(text="因为 BD 平分角 ABC，所以角 ABD 等于三十五度，也就等于角 ACB。") as tracker:
            self.play_steps_evenly([Write(panel1[2])], tracker)

        with self.voiceover(text="又因为角 BAD 和角 BAC 是公共角，所以三角形 ABD 相似于三角形 ACB。") as tracker:
            self.play_steps_evenly([Write(panel1[3]), FadeIn(region_ABD), FadeIn(region_ACB)], tracker)

        # 清理第一屏，准备第二屏
        self.play(FadeOut(panel1), FadeOut(region_ABD), FadeOut(region_ACB))

        # 第二屏
        panel2 = self.make_step_panel([
            self.make_ctex_row(r"设 $AB=c$，$AC=b$，$BC=a$，$AD=y$，$BD=x$"),
            self.make_proof_because(r"\triangle ABD \sim \triangle ACB"),
            self.make_proof_therefore(r"\frac{AB}{AC} = \frac{AD}{AB} = \frac{BD}{BC} \implies c^2 = by, \frac{x}{a} = \frac{c}{b}"),
            self.make_ctex_row(r"由(1)知 $BD=DC$，得 $b = y + x$")
        ])

        with self.voiceover(text="为了计算方便，我们设 AB 为 c，AC 为 b，BC 为 a，AD 为 y，BD 为 x。") as tracker:
            self.play_steps_evenly([Write(panel2[0])], tracker)

        with self.voiceover(text="由相似三角形对应边成比例，得到 c 的平方等于 b 乘以 y，且 x 比 a 等于 c 比 b。") as tracker:
            self.play_steps_evenly([Write(panel2[1]), Write(panel2[2])], tracker)

        with self.voiceover(text="根据第一问的结论，BD 等于 DC，所以 b 等于 y 加 x。") as tracker:
            self.play_steps_evenly([Write(panel2[3])], tracker)

        # 清理第二屏，准备第三屏
        self.play(FadeOut(panel2))

        # 第三屏
        panel3 = self.make_step_panel([
            self.make_ctex_row(r"得 $x = b - y = b - \frac{c^2}{b} = \frac{b^2 - c^2}{b}$"),
            self.make_ctex_row(r"由 $\frac{x}{a} = \frac{c}{b}$，得 $a = \frac{b^2 - c^2}{c}$，即 $ac = (b-c)(b+c)$"),
            self.make_proof_because(r"AE \text{ 是 } \angle BAC \text{ 的外角平分线}"),
            self.make_proof_therefore(r"\frac{BE}{CE} = \frac{AB}{AC} = \frac{c}{b}")
        ])

        with self.voiceover(text="由此可以表示出 x 等于 b 减去 y，代入后等于 b 分之 b 的平方减去 c 的平方。") as tracker:
            self.play_steps_evenly([Write(panel3[0])], tracker)

        with self.voiceover(text="再结合比例式，可以求得 a 等于 c 分之 b 的平方减去 c 的平方，即 a 乘以 c 等于 b 减 c 乘以 b 加 c。") as tracker:
            self.play_steps_evenly([Write(panel3[1])], tracker)

        with self.voiceover(text="因为 AE 是角 BAC 的外角平分线，由外角平分线定理，BE 比 CE 等于 AB 比 AC，等于 c 比 b。") as tracker:
            self.play_steps_evenly([Write(panel3[2]), Write(panel3[3])], tracker)

        # 清理第三屏，准备第四屏
        self.play(FadeOut(panel3))

        # 第四屏
        panel4 = self.make_step_panel([
            self.make_ctex_row(r"又 $CE = a + BE$，得 $\frac{BE}{a+BE} = \frac{c}{b}$"),
            self.make_ctex_row(r"\implies (b-c)BE = ac"),
            self.make_ctex_row(r"\text{代入 } ac = (b-c)(b+c) \text{ 得 } BE = b+c = AC+AB"),
            self.make_ctex_row(r"\text{因为 } AC = AD+BD\text{，所以 } BE = AB+AD+BD")
        ])

        with self.voiceover(text="因为 CE 等于 a 加 BE，代入比例式中，整理可以得到 b 减 c 乘以 BE 等于 a 乘以 c。") as tracker:
            self.play_steps_evenly([Write(panel4[0]), Write(panel4[1])], tracker)

        with self.voiceover(text="把前面得到的 ac 表达式代入，约分后就得到 BE 等于 b 加 c，也就是 AC 加 AB。") as tracker:
            self.play_steps_evenly([Write(panel4[2])], tracker)

        with self.voiceover(text="最后，因为 AC 等于 AD 加 BD，所以 BE 就等于 AB 加 AD 加 BD。") as tracker:
            self.play_steps_evenly([Write(panel4[3])], tracker)

        # ==================== 结论收束与技巧总结 ====================
        with self.voiceover(text="所以，第二问的等式不再成立，正确的线段关系是 BE 等于 AB 加 AD 加 BD。") as tracker:
            self.play_steps_evenly([
                panel4[3].animate.set_color(RESULT_COLOR),
                self.reemphasize_stroke(edge_BE_real)
            ], tracker)

        with self.voiceover(text="在解决外角平分线问题时，大家要先确认点在线段上还是延长线上，熟练运用外角平分线定理，并且注意线段的位置关系不要写反。") as tracker:
            self.wait(2)
