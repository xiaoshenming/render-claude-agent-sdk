```python
# PROBLEM_TEXT: 1. 如下图，等边三角形 ABC 中，D、E 分别在 AB、AC 边上，且 AD = CE，BE、CD 交于 P 点，则图中 60° 的角共有（ ）
# FIGURE_FACTS: 等边三角形 ABC, D 在 AB 上, E 在 AC 上, AD = CE, BE 与 CD 交于 P

from manim import *
import numpy as np

class EquilateralTriangleProblem(Scene):
    def construct(self):
        # Define vertices of the equilateral triangle ABC
        # Side length = 5
        side_length = 5.0
        height = side_length * np.sqrt(3) / 2
        
        B = np.array([-side_length / 2, -height / 3, 0])
        C = np.array([side_length / 2, -height / 3, 0])
        A = np.array([0, 2 * height / 3, 0])
        
        # D is on AB, E is on AC such that AD = CE
        # Let's set AD = 0.35 * AB
        r = 0.35
        D = A + r * (B - A)
        E = C + r * (A - C)
        
        # Calculate intersection P of CD and BE
        # Line 1: C -> D, Line 2: B -> E
        def intersect(p1, p2, p3, p4):
            x1, y1, _ = p1
            x2, y2, _ = p2
            x3, y3, _ = p3
            x4, y4, _ = p4
            denom = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
            if abs(denom) < 1e-9:
                return None
            px = ((x1*y2 - y1*x2)*(x3 - x4) - (x1 - x2)*(x3*y4 - y3*x4)) / denom
            py = ((x1*y2 - y1*x2)*(y3 - y4) - (y1 - y2)*(x3*y4 - y3*x4)) / denom
            return np.array([px, py, 0])
            
        P = intersect(C, D, B, E)
        
        # Create geometry
        triangle = Polygon(A, B, C, color=WHITE, stroke_width=2)
        line_CD = Line(C, D, color=WHITE, stroke_width=2)
        line_BE = Line(B, E, color=WHITE, stroke_width=2)
        
        # Labels
        label_A = MathTex("A").next_to(A, UP, buff=0.15)
        label_B = MathTex("B").next_to(B, DL, buff=0.15)
        label_C = MathTex("C").next_to(C, DR, buff=0.15)
        label_D = MathTex("D").next_to(D, LEFT, buff=0.15)
        label_E = MathTex("E").next_to(E, RIGHT, buff=0.15)
        label_P = MathTex("P").next_to(P, DOWN, buff=0.15)
        
        # Group and display
        geometry = VGroup(triangle, line_CD, line_BE, label_A, label_B, label_C, label_D, label_E, label_P)
        self.add(geometry)
```
