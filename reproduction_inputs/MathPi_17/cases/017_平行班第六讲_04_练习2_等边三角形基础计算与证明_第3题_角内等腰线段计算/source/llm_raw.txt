class Solution(BaseConfigScene):
    def construct(self):
        # 1. 原始几何坐标定义
        O_raw = np.array([0, 0, 0])
        A_raw = np.array([6, 6 * np.sqrt(3), 0])
        B_raw = np.array([9, 0, 0])
        P_raw = np.array([5, 5 * np.sqrt(3), 0])
        M_raw = np.array([4, 0, 0])
        N_raw = np.array([6, 0, 0])

        # 2. 基础图形创建
        line_OA = Line(O_raw, A_raw, color=BASE_COLOR)
        line_OB = Line(O_raw, B_raw, color=BASE_COLOR)
        line_PM = Line(P_raw, M_raw, color=BASE_COLOR)
        line_PN = Line(P_raw, N_raw, color=BASE_COLOR)
        diagram = VGroup(line_OA, line_OB, line_PM, line_PN)

        # 3. 首次适配左侧
        self.fit_left(diagram)

        # 4. 获取适配后的实时坐标
        O = line_OA.get_start()
        A = line_OA.get_end()
        B = line_OB.get_end()
        P = line_PM.get_start()
        M = line_PM.get_end()
        N = line_PN.get_end()
        H = np.array([P[0], O[1], 0])

        # 5. 用实时坐标创建标签、辅助线、角标、刻痕与区域
        lbl_O = MathTex("O", color=TEXT_COLOR).next_to(O, DL, buff=0.15)
        lbl_A = MathTex("A", color=TEXT_COLOR).next_to(A, UR, buff=0.15)
        lbl_B = MathTex("B", color=TEXT_COLOR).next_to(B, DR, buff=0.15)
        lbl_P = MathTex("P", color=TEXT_COLOR).next_to(P, UL, buff=0.15)
        lbl_M = MathTex("M", color=TEXT_COLOR).next_to(M, DOWN, buff=0.15)
        lbl_N = MathTex("N", color=TEXT_COLOR).next_to(N, DOWN, buff=0.15)
        lbl_H = MathTex("H", color=TEXT_COLOR).next_to(H, DOWN, buff=0.15)

        aux_PH = DashedLine(P, H, color=DRAW_COLOR)
        arc_AOB = self.make_angle_mark(O, A, B, radius=0.4, color=KNOWN_COLOR)
        right_PHO = self.make_right_angle_mark_at(H, P, O, size=0.2, color=DRAW_COLOR)
        tick_PM = self.make_tick_mark(line_PM, n=1, color=KNOWN_COLOR)
        tick_PN = self.make_tick_mark(line_PN, n=1, color=KNOWN_COLOR)
        
        region_PMN = self.make_region_fill(P, M, N, color=PAIR_A_COLOR, opacity=0.18)
        region_OPH = self.make_region_fill(O, P, H, color=PAIR_B_COLOR, opacity=0.18)

        # 6. 整体适配
        geometry_all = VGroup(
            diagram, aux_PH, arc_AOB, right_PHO, tick_PM, tick_PN, 
            region_PMN, region_OPH, lbl_O, lbl_A, lbl_B, lbl_P, lbl_M, lbl_N, lbl_H
        )
        self.fit_geometry_group(geometry_all)

        # 7. 隐藏后续才出现的元素
        aux_PH.set_opacity(0)
        right_PHO.set_opacity(0)
        lbl_H.set_opacity(0)
        tick_PM.set_opacity(0)
        tick_PN.set_opacity(0)
        region_PMN.set_opacity(0)
        region_OPH.set_opacity(0)
        arc_AOB.set_opacity(0)

        # 8. 动画与口播结合
        # --- 读题展示阶段 ---
        with self.voiceover(text="同学们好，我们一起来看这道几何题。") as tracker:
            self.play(
                FadeIn(diagram), 
                FadeIn(lbl_O), 
                FadeIn(lbl_A), 
                FadeIn(lbl_B), 
                FadeIn(lbl_P), 
                FadeIn(lbl_M), 
                FadeIn(lbl_N)
            )

        with self.voiceover(text="已知角AOB等于六十度，点P在角AOB内部。") as tracker:
            self.play(FadeIn(arc_AOB))
            self.highlight_angle(arc_AOB)

        with self.voiceover(text="并且线段PM等于PN，线段OP等于十，MN等于二。") as tracker:
            self.play(
                FadeIn(tick_PM), 
                FadeIn(tick_PN)
            )
            self.highlight_edge(line_PM)
            self.highlight_edge(line_PN)

        # 目标面板
        goal_row = self.make_ctex_row(r"求 $OM$ 的长")
        goal_panel = self.make_question_goal_panel(goal_row)
        with self.voiceover(text="我们的目标是求线段OM的长度。") as tracker:
            line_OM_temp = Line(O, M, color=FOCUS_COLOR, stroke_width=4)
            self.play(
                FadeIn(goal_panel),
                Create(line_OM_temp)
            )
            self.play(FadeOut(line_OM_temp))

        # --- 思路切入阶段 ---
        with self.voiceover(text="看到等腰三角形PMN，我们很自然地想到三线合一的性质。") as tracker:
            self.play(FadeIn(region_PMN))

        with self.voiceover(text="而面对六十度的角，我们则想构造含有三十度角的直角三角形。") as tracker:
            self.play(FadeOut(region_PMN))
            self.reemphasize_element(arc_AOB)

        with self.voiceover(text="因此，我们过点P作PH垂直于OB，垂足为H。") as tracker:
            self.play(
                aux_PH.animate.set_opacity(1),
                right_PHO.animate.set_opacity(1),
                FadeIn(lbl_H)
            )

        # --- 具体推导过程 ---
        # 第一屏板书
        row1_1 = self.make_ctex_row(r"过点 $P$ 作 $PH \perp OB$ 于 $H$")
        row1_2 = self.make_proof_because(r"PM=PN")
        row1_3 = self.make_proof_therefore(r"MH=NH=\frac{1}{2}MN")
        row1_4 = self.make_ctex_row(r"因为 $MN=2$，所以 $MH=1$")
        panel1 = self.make_step_panel([row1_1, row1_2, row1_3, row1_4])

        with self.voiceover(text="我们开始书写步骤。第一步，作辅助线PH垂直于OB。") as tracker:
            self.play(Write(row1_1))

        with self.voiceover(text="因为PM等于PN，且PH垂直于MN。") as tracker:
            self.play(Write(row1_2))
            self.reemphasize_stroke(line_PM)
            self.reemphasize_stroke(line_PN)

        with self.voiceover(text="根据等腰三角形三线合一，H是MN的中点，所以MH等于NH等于二分之一MN。") as tracker:
            self.play(Write(row1_3))

        with self.voiceover(text="已知MN等于二，因此MH的长度就等于一。") as tracker:
            self.play(Write(row1_4))

        # 第二屏板书
        row2_1 = self.make_ctex_row(r"在 $\text{Rt}\triangle OPH$ 中，$\angle OHP=90^\circ$")
        row2_2 = self.make_ctex_row(r"因为 $\angle AOB=60^\circ$，所以 $\angle OPH=30^\circ$")
        row2_3 = self.make_proof_therefore(r"OH=\frac{1}{2}OP")
        row2_4 = self.make_ctex_row(r"因为 $OP=10$，所以 $OH=5$")
        panel2 = self.make_step_panel([row2_1, row2_2, row2_3, row2_4])

        with self.voiceover(text="第二步，我们把视线移到直角三角形OPH中。") as tracker:
            self.play(
                FadeOut(panel1),
                FadeIn(region_OPH),
                Write(row2_1)
            )

        with self.voiceover(text="因为角AOB等于六十度，所以角OPH等于九十度减去六十度，等于三十度。") as tracker:
            self.play(Write(row2_2))

        with self.voiceover(text="根据直角三角形中，三十度角所对的直角边等于斜边的一半。") as tracker:
            line_OH_temp = Line(O, H, color=FOCUS_COLOR, stroke_width=4)
            line_OP_temp = Line(O, P, color=FOCUS_COLOR, stroke_width=4)
            self.play(
                Write(row2_3),
                Create(line_OH_temp),
                Create(line_OP_temp)
            )
            self.play(FadeOut(line_OH_temp), FadeOut(line_OP_temp))

        with self.voiceover(text="已知OP等于十，所以直角边OH的长度就等于五。") as tracker:
            self.play(Write(row2_4))

        # 第三屏板书
        row3_1 = self.make_ctex_row(r"由线段和差关系：")
        row3_2 = self.make_ctex_row(r"$OM = OH - MH$")
        row3_3 = self.make_ctex_row(r"代入数据得：$OM = 5 - 1 = 4$")
        panel3 = self.make_step_panel([row3_1, row3_2, row3_3])

        with self.voiceover(text="第三步，我们来计算目标线段OM的长度。") as tracker:
            self.play(
                FadeOut(panel2),
                FadeOut(region_OPH),
                Write(row3_1)
            )

        with self.voiceover(text="由线段和差关系，OM等于OH减去MH。") as tracker:
            self.play(Write(row3_2))

        with self.voiceover(text="代入我们求得的数值，得到OM等于五减去一，结果为四。") as tracker:
            self.play(Write(row3_3))

        # --- 结论与技巧总结 ---
        with self.voiceover(text="这道题的关键在于通过作高线PH，既利用了等腰三角形的三线合一，又构造了含三十度角的直角三角形。") as tracker:
            self.reemphasize_stroke(aux_PH)

        with self.voiceover(text="从而顺利将未知线段转化为已知线段的和差来求解。你学会了吗？") as tracker:
            line_OM_final = Line(O, M, color=RESULT_COLOR, stroke_width=5)
            self.play(Create(line_OM_final))
            self.play(FadeOut(line_OM_final))
