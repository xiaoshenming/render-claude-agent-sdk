class Solution(BaseConfigScene):
    def construct(self):
        # 1. 定义初始顶点坐标（静态坐标）
        A = np.array([-1.8, 1.7, 0])
        B = np.array([-2.3, -1.8, 0])
        C = np.array([2.3, -1.8, 0])
        D = np.array([1.7, 1.8, 0])
        E = np.array([-1.0, -1.8, 0])
        F = np.array([1.94, 0.36, 0])
        
        # 2. 创建初始黑色原图线段
        ab = Line(A, B, stroke_width=3, color=BASE_COLOR)
        bc = Line(B, C, stroke_width=3, color=WHITE)
        cd = Line(C, D, stroke_width=3, color=WHITE)
        da = Line(D, A, stroke_width=3, color=BASE_COLOR)
        ae = Line(A, E, stroke_width=3, color=WHITE)
        af = Line(A, F, stroke_width=3, color=WHITE)
        ef = Line(E, F, stroke_width=3, color=WHITE)
        
        diagram = VGroup(ab, bc, cd, da, ae, af, ef)
        
        # 3. 首次适配左侧图形区
        self.fit_left(diagram)
        
        # 4. 获取适配后的实时坐标
        pt_A = ab.get_start()
        pt_B = ab.get_end()
        pt_C = bc.get_end()
        pt_D = da.get_start()
        pt_E = ae.get_end()
        pt_F = af.get_end()
        
        # 计算 G 的实时坐标 (延长 CB 到 G，使 BG = DF)
        dir_CB = (pt_B - pt_C) / np.linalg.norm(pt_B - pt_C)
        len_DF = np.linalg.norm(pt_D - pt_F)
        pt_G = pt_B + dir_CB * len_DF
        
        # 5. 用实时坐标创建辅助线、标签、刻痕和区域色块
        aux_BG = Line(pt_B, pt_G, stroke_width=3, color=DRAW_COLOR)
        aux_AG = Line(pt_A, pt_G, stroke_width=3, color=DRAW_COLOR)
        
        lbl_A = MathTex("A", color=TEXT_COLOR).next_to(pt_A, UP, buff=0.15)
        lbl_B = MathTex("B", color=TEXT_COLOR).next_to(pt_B, DOWN + RIGHT, buff=0.15)
        lbl_C = MathTex("C", color=TEXT_COLOR).next_to(pt_C, DOWN + RIGHT, buff=0.15)
        lbl_D = MathTex("D", color=TEXT_COLOR).next_to(pt_D, UP + RIGHT, buff=0.15)
        lbl_E = MathTex("E", color=TEXT_COLOR).next_to(pt_E, DOWN, buff=0.15)
        lbl_F = MathTex("F", color=TEXT_COLOR).next_to(pt_F, RIGHT, buff=0.15)
        lbl_G = MathTex("G", color=TEXT_COLOR).next_to(pt_G, DOWN + LEFT, buff=0.15)
        
        tick_AB = self.make_tick_mark(ab, n=1, color=KNOWN_COLOR)
        tick_AD = self.make_tick_mark(da, n=1, color=KNOWN_COLOR)
        tick_BG = self.make_tick_mark(aux_BG, n=2, color=KNOWN_COLOR)
        
        line_DF = Line(pt_D, pt_F, stroke_width=3, color=BASE_COLOR)
        tick_DF = self.make_tick_mark(line_DF, n=2, color=KNOWN_COLOR)
        
        region_ABG = self.make_region_fill(pt_A, pt_B, pt_G, color=PAIR_A_COLOR, opacity=0.18)
        region_ADF = self.make_region_fill(pt_A, pt_D, pt_F, color=PAIR_B_COLOR, opacity=0.18)
        region_GAE = self.make_region_fill(pt_A, pt_G, pt_E, color=PAIR_A_COLOR, opacity=0.18)
        region_FAE = self.make_region_fill(pt_A, pt_F, pt_E, color=PAIR_B_COLOR, opacity=0.18)
        
        # 6. 整体适配
        geometry_all = VGroup(
            diagram, aux_BG, aux_AG, lbl_A, lbl_B, lbl_C, lbl_D, lbl_E, lbl_F, lbl_G,
            tick_AB, tick_AD, tick_BG, tick_DF, region_ABG, region_ADF, region_GAE, region_FAE
        )
        self.fit_geometry_group(geometry_all)
        
        # 7. 初始显示原图和基础标签
        self.add(diagram, lbl_A, lbl_B, lbl_C, lbl_D, lbl_E, lbl_F)
        
        # ==================== 读题展示阶段 ====================
        with self.voiceover(text="同学们好，我们一起来看这道题。在四边形 ABCD 中，已知 AB 等于 AD。") as tracker:
            self.play_after_speech([
                self.highlight_edge(ab),
                self.highlight_edge(da),
                Create(tick_AB),
                Create(tick_AD)
            ], tracker)
            
        with self.voiceover(text="并且对角互补，也就是角 B 加上角 D 等于一百八十度。") as tracker:
            self.play_after_speech([
                self.highlight_point(lbl_B),
                self.highlight_point(lbl_D)
            ], tracker)
            
        with self.voiceover(text="点 E 和点 F 分别是边 BC 和 CD 上的点。") as tracker:
            self.play_after_speech([
                self.highlight_point(lbl_E),
                self.highlight_point(lbl_F)
            ], tracker)
            
        goal_panel = self.make_question_goal_panel(self.make_ctex_row(r"求证：$BE+FD=EF$"))
        with self.voiceover(text="第一问，如果角 EAF 等于二分之一的角 BAD，求证 BE 加上 FD 等于 EF。") as tracker:
            self.play_after_speech([FadeIn(goal_panel)], tracker)
            
        # ==================== 思路切入阶段 ====================
        with self.voiceover(text="直接在四边形里推导线段和角的关系比较困难，我们可以利用对称性进行旋转拼接。") as tracker:
            pass
            
        with self.voiceover(text="通过延长线段，把分散的线段 FD 和 BE 转移到同一条直线上来证明。") as tracker:
            pass
            
        # ==================== 第一问证明阶段 ====================
        # 创建第一屏板书行
        row1 = self.make_proof_because(r"\angle ABC + \angle D = 180^\circ, \angle ABC + \angle ABG = 180^\circ")
        row2 = self.make_proof_therefore(r"\angle ABG = \angle D")
        row3 = self.make_proof_because(r"\begin{cases}AB=AD \\ \angle ABG=\angle D \\ BG=DF\end{cases}")
        row4 = self.make_proof_therefore(r"\triangle ABG \cong \triangle ADF\ (SAS)")
        panel1 = self.make_step_panel([row1, row2, row3, row4])
        
        with self.voiceover(text="首先，我们延长 CB 到点 G，使 BG 等于 DF，并连接 AG。") as tracker:
            self.play_steps_evenly([
                Create(aux_BG),
                Create(aux_AG),
                FadeIn(lbl_G),
                Create(tick_BG),
                Create(tick_DF)
            ], tracker)
            
        with self.voiceover(text="因为角 ABC 与角 D 互补，且角 ABC 与角 ABG 互补。") as tracker:
            self.play_steps_evenly([Write(row1)], tracker)
            
        with self.voiceover(text="所以角 ABG 等于角 D。") as tracker:
            self.play_steps_evenly([Write(row2)], tracker)
            
        with self.voiceover(text="现在我们来看三角形 ABG 和三角形 ADF。") as tracker:
            self.play_steps_evenly([
                FadeIn(region_ABG),
                FadeIn(region_ADF)
            ], tracker)
            
        with self.voiceover(text="在这两个三角形中，有 AB 等于 AD，角 ABG 等于角 D，且 BG 等于 DF。") as tracker:
            self.play_steps_evenly([
                Write(row3),
                self.reemphasize_equal_segments(ab, da),
                self.reemphasize_equal_segments(aux_BG, line_DF)
            ], tracker)
            
        with self.voiceover(text="所以这两个三角形全等，判定依据是边角边。") as tracker:
            self.play_steps_evenly([Write(row4)], tracker)
            
        # 清屏第一屏板书
        self.play(FadeOut(row1), FadeOut(row2), FadeOut(row3), FadeOut(row4))
        
        # 创建第二屏板书行
        row5 = self.make_proof_therefore(r"AG=AF")
        row6 = self.make_proof_therefore(r"\angle BAG = \angle DAF")
        row7 = self.make_proof_because(r"\angle EAF = \frac{1}{2}\angle BAD")
        row8 = self.make_proof_therefore(r"\angle BAE + \angle DAF = \frac{1}{2}\angle BAD")
        panel2 = self.make_step_panel([row5, row6, row7, row8])
        
        with self.voiceover(text="由全等三角形的性质，我们可以得到对应边 AG 等于 AF。") as tracker:
            self.play_steps_evenly([Write(row5)], tracker)
            
        with self.voiceover(text="并且对应角，角 BAG 等于角 DAF。") as tracker:
            self.play_steps_evenly([Write(row6)], tracker)
            
        with self.voiceover(text="因为已知角 EAF 等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([Write(row7)], tracker)
            
        with self.voiceover(text="所以角 BAE 加上角 DAF 也等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([Write(row8)], tracker)
            
        # 清屏第二屏板书
        self.play(FadeOut(row5), FadeOut(row6), FadeOut(row7), FadeOut(row8))
        
        # 创建第三屏板书行
        row9 = self.make_proof_therefore(r"\angle BAE + \angle BAG = \frac{1}{2}\angle BAD")
        row10 = self.make_proof_therefore(r"\angle GAE = \frac{1}{2}\angle BAD")
        row11 = self.make_proof_therefore(r"\angle GAE = \angle EAF")
        row12 = self.make_proof_because(r"\begin{cases}AG=AF \\ \angle GAE=\angle EAF \\ AE=AE\end{cases}")
        panel3 = self.make_step_panel([row9, row10, row11, row12])
        
        with self.voiceover(text="把角 DAF 替换为角 BAG，得到角 BAE 加上角 BAG 等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([Write(row9)], tracker)
            
        with self.voiceover(text="也就是角 GAE 等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([Write(row10)], tracker)
            
        with self.voiceover(text="从而我们证明了角 GAE 等于角 EAF。") as tracker:
            self.play_steps_evenly([Write(row11)], tracker)
            
        with self.voiceover(text="接下来，我们来观察三角形 GAE 和三角形 FAE。") as tracker:
            self.play_steps_evenly([
                FadeOut(region_ABG),
                FadeOut(region_ADF),
                FadeIn(region_GAE),
                FadeIn(region_FAE)
            ], tracker)
            
        with self.voiceover(text="在这两个三角形中，有 AG 等于 AF，角 GAE 等于角 EAF，且公共边 AE 等于 AE。") as tracker:
            self.play_steps_evenly([Write(row12)], tracker)
            
        # 清屏第三屏板书
        self.play(FadeOut(row9), FadeOut(row10), FadeOut(row11), FadeOut(row12))
        
        # 创建第四屏板书行
        row13 = self.make_proof_therefore(r"\triangle GAE \cong \triangle FAE\ (SAS)")
        row14 = self.make_proof_therefore(r"GE=EF")
        row15 = self.make_proof_because(r"GE = BE + BG = BE + DF")
        row16 = self.make_proof_therefore(r"BE+FD=EF")
        panel4 = self.make_step_panel([row13, row14, row15, row16])
        
        with self.voiceover(text="所以三角形 GAE 全等于三角形 FAE。") as tracker:
            self.play_steps_evenly([Write(row13)], tracker)
            
        with self.voiceover(text="从而对应边 GE 等于 EF。") as tracker:
            self.play_steps_evenly([Write(row14)], tracker)
            
        with self.voiceover(text="因为 GE 等于 BE 加上 BG，而 BG 等于 DF。") as tracker:
            self.play_steps_evenly([Write(row15)], tracker)
            
        with self.voiceover(text="所以我们成功证明了 BE 加上 FD 等于 EF。") as tracker:
            self.play_steps_evenly([Write(row16)], tracker)
            
        # 第一问结束，清屏
        self.play(
            FadeOut(row13), FadeOut(row14), FadeOut(row15), FadeOut(row16),
            FadeOut(goal_panel), FadeOut(region_GAE), FadeOut(region_FAE)
        )
        
        # ==================== 第二问证明阶段 ====================
        goal_panel2 = self.make_question_goal_panel(self.make_ctex_row(r"求证：$\angle EAF=\frac{1}{2}\angle BAD$"))
        
        with self.voiceover(text="第二问是第一问的逆命题。已知 BE 加上 FD 等于 EF，求证角 EAF 等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([FadeIn(goal_panel2)], tracker)
            
        with self.voiceover(text="我们同样延长 CB 到点 G，使 BG 等于 DF，并连接 AG。") as tracker:
            self.play_steps_evenly([
                self.reemphasize_stroke(aux_BG),
                self.reemphasize_stroke(aux_AG)
            ], tracker)
            
        # 创建第五屏板书行
        row17 = self.make_proof_therefore(r"\triangle ABG \cong \triangle ADF\ (SAS)")
        row18 = self.make_proof_therefore(r"AG=AF，\angle BAG = \angle DAF")
        row19 = self.make_proof_because(r"EF=BE+FD，BG=DF")
        row20 = self.make_proof_therefore(r"EF = BE + BG = GE")
        panel5 = self.make_step_panel([row17, row18, row19, row20])
        
        with self.voiceover(text="同理可以证明三角形 ABG 全等于三角形 ADF。") as tracker:
            self.play_steps_evenly([
                Write(row17),
                FadeIn(region_ABG),
                FadeIn(region_ADF)
            ], tracker)
            
        with self.voiceover(text="从而得到 AG 等于 AF，且角 BAG 等于角 DAF。") as tracker:
            self.play_steps_evenly([Write(row18)], tracker)
            
        with self.voiceover(text="因为已知 EF 等于 BE 加上 FD，且 BG 等于 DF。") as tracker:
            self.play_steps_evenly([Write(row19)], tracker)
            
        with self.voiceover(text="所以 EF 等于 BE 加上 BG，也就是 GE。") as tracker:
            self.play_steps_evenly([Write(row20)], tracker)
            
        # 清屏第五屏板书
        self.play(FadeOut(row17), FadeOut(row18), FadeOut(row19), FadeOut(row20))
        
        # 创建第六屏板书行
        row21 = self.make_proof_because(r"\begin{cases}AG=AF \\ GE=EF \\ AE=AE\end{cases}")
        row22 = self.make_proof_therefore(r"\triangle GAE \cong \triangle FAE\ (SSS)")
        row23 = self.make_proof_therefore(r"\angle GAE = \angle EAF")
        row24 = self.make_proof_because(r"\angle GAE = \angle BAE + \angle BAG = \angle BAE + \angle DAF")
        panel6 = self.make_step_panel([row21, row22, row23, row24])
        
        with self.voiceover(text="现在我们再次观察三角形 GAE 和三角形 FAE。") as tracker:
            self.play_steps_evenly([
                FadeOut(region_ABG),
                FadeOut(region_ADF),
                FadeIn(region_GAE),
                FadeIn(region_FAE)
            ], tracker)
            
        with self.voiceover(text="在这两个三角形中，有 AG 等于 AF，GE 等于 EF，且公共边 AE 等于 AE。") as tracker:
            self.play_steps_evenly([Write(row21)], tracker)
            
        with self.voiceover(text="所以这两个三角形全等，判定依据是边边边。") as tracker:
            self.play_steps_evenly([Write(row22)], tracker)
            
        with self.voiceover(text="从而对应角，角 GAE 等于角 EAF。") as tracker:
            self.play_steps_evenly([Write(row23)], tracker)
            
        with self.voiceover(text="因为角 GAE 等于角 BAE 加上角 BAG，也就是角 BAE 加上角 DAF。") as tracker:
            self.play_steps_evenly([Write(row24)], tracker)
            
        # 清屏第六屏板书
        self.play(FadeOut(row21), FadeOut(row22), FadeOut(row23), FadeOut(row24))
        
        # 创建第七屏板书行
        row25 = self.make_proof_therefore(r"\angle EAF = \angle BAE + \angle DAF")
        row26 = self.make_proof_therefore(r"\angle BAD = \angle BAE + \angle EAF + \angle DAF = 2\angle EAF")
        row27 = self.make_proof_therefore(r"\angle EAF = \frac{1}{2}\angle BAD")
        panel7 = self.make_step_panel([row25, row26, row27])
        
        with self.voiceover(text="所以角 EAF 等于角 BAE 加上角 DAF。") as tracker:
            self.play_steps_evenly([Write(row25)], tracker)
            
        with self.voiceover(text="因此角 BAD 等于角 BAE 加上角 EAF 加上角 DAF，等于两倍的角 EAF。") as tracker:
            self.play_steps_evenly([Write(row26)], tracker)
            
        with self.voiceover(text="这就证明了角 EAF 等于二分之一的角 BAD。") as tracker:
            self.play_steps_evenly([Write(row27)], tracker)
            
        # ==================== 技巧总结阶段 ====================
        with self.voiceover(text="这道题非常经典，我们通过旋转拼接构造全等三角形，成功实现了线段和角度的转移。") as tracker:
            pass
            
        with self.voiceover(text="在解决几何综合题时，灵活运用辅助线进行图形变换是突破难点的关键。") as tracker:
            pass
            
        # 结束淡出
        self.play(
            FadeOut(row25), FadeOut(row26), FadeOut(row27),
            FadeOut(goal_panel2), FadeOut(region_GAE), FadeOut(region_FAE)
        )
        self.wait(1)
